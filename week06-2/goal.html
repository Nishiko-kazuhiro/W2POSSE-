const price = document.querySelector('#price').value; // "1200"（文字列）
const qty   = document.querySelector('#qty').value;   // "3"（文字列）
const total = price * qty;   // 3600（偶然動く）
const tax   = total * 0.1;
const result = total + tax;  // 3960（数値）
console.log(result + "円");  // "3960円"

上のコードで、total + tax の計算は正しく動きますか？なぜですか？
